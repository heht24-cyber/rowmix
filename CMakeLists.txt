cmake_minimum_required(VERSION 3.16)
project(RowMix VERSION 0.1.0 LANGUAGES C CXX)
option(ROWMIX_PORTABLE_ONLY "Build only the platform-independent scalar kernel" OFF)
add_library(rowmix SHARED native/rowmix.cpp native/kernels.cpp)
target_include_directories(rowmix PUBLIC native)
target_compile_features(rowmix PRIVATE cxx_std_17)
set_target_properties(rowmix PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(ROWMIX_PORTABLE_ONLY)
  target_compile_definitions(rowmix PRIVATE ROWMIX_PORTABLE_ONLY)
endif()
if(MSVC)
  target_compile_options(rowmix PRIVATE /fp:strict /W4)
else()
  target_compile_options(rowmix PRIVATE -ffp-contract=off -Wall -Wextra)
endif()
add_executable(rowmix-run native/runner.cpp)
target_compile_features(rowmix-run PRIVATE cxx_std_17)
target_link_libraries(rowmix-run PRIVATE rowmix)
add_executable(rowmix-embed examples/embed.c)
target_link_libraries(rowmix-embed PRIVATE rowmix)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME native-sample COMMAND rowmix-embed
    ${PROJECT_SOURCE_DIR}/examples/digits/mixed.rmx ${PROJECT_SOURCE_DIR}/examples/digits/sample.f32)
endif()
install(TARGETS rowmix rowmix-run)
install(FILES native/rowmix.h DESTINATION include)
